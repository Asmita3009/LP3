# ============================================
# UBER FARE PREDICTION
# Linear Regression vs Random Forest Regression
# ============================================

# 1. Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error


# ============================================
# 2. Load Dataset
# ============================================

df = pd.read_csv("uber.csv")

print("Dataset Shape:", df.shape)
print(df.head())


# ============================================
# 3. Pre-processing
# ============================================

# Remove unnecessary columns
df.drop(
    columns=['Unnamed: 0', 'key'],
    errors='ignore',
    inplace=True
)

# Remove missing values
df.dropna(inplace=True)

# Convert datetime column
df['pickup_datetime'] = pd.to_datetime(
    df['pickup_datetime']
)

# Extract date/time features
df['hour'] = df['pickup_datetime'].dt.hour
df['day'] = df['pickup_datetime'].dt.day
df['month'] = df['pickup_datetime'].dt.month
df['year'] = df['pickup_datetime'].dt.year
df['day_of_week'] = df['pickup_datetime'].dt.dayofweek

# Remove original datetime column
df.drop(
    columns=['pickup_datetime'],
    inplace=True
)

# Remove invalid values
df = df[df['fare_amount'] > 0]
df = df[df['passenger_count'] > 0]


# ============================================
# 4. Identify Outliers
# ============================================

Q1 = df['fare_amount'].quantile(0.25)
Q3 = df['fare_amount'].quantile(0.75)

IQR = Q3 - Q1

lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

print("\nFare Outlier Limits:")
print("Lower Bound:", lower_bound)
print("Upper Bound:", upper_bound)

# Number of outliers
outliers = df[
    (df['fare_amount'] < lower_bound) |
    (df['fare_amount'] > upper_bound)
]

print("Number of outliers:", len(outliers))


# Boxplot before removing outliers
plt.figure(figsize=(8, 5))
sns.boxplot(x=df['fare_amount'])
plt.title("Fare Amount - Before Outlier Removal")
plt.show()


# Remove outliers
df = df[
    (df['fare_amount'] >= lower_bound) &
    (df['fare_amount'] <= upper_bound)
]


# Boxplot after removing outliers
plt.figure(figsize=(8, 5))
sns.boxplot(x=df['fare_amount'])
plt.title("Fare Amount - After Outlier Removal")
plt.show()


# ============================================
# 5. Calculate Trip Distance
# ============================================

# Convert latitude and longitude to radians
lon1 = np.radians(df['pickup_longitude'])
lat1 = np.radians(df['pickup_latitude'])

lon2 = np.radians(df['dropoff_longitude'])
lat2 = np.radians(df['dropoff_latitude'])

# Difference in coordinates
dlon = lon2 - lon1
dlat = lat2 - lat1

# Haversine formula
a = (
    np.sin(dlat / 2) ** 2
    + np.cos(lat1)
    * np.cos(lat2)
    * np.sin(dlon / 2) ** 2
)

c = 2 * np.arcsin(np.sqrt(a))

# Earth radius in kilometres
df['distance_km'] = 6371 * c

# Remove zero and unrealistic distances
df = df[
    (df['distance_km'] > 0) &
    (df['distance_km'] < 100)
]

print("\nDistance calculated successfully.")
print(df[['distance_km', 'fare_amount']].head())


# ============================================
# 6. Check Correlation
# ============================================

correlation = df.corr()

print("\nCorrelation Matrix:")
print(correlation)

plt.figure(figsize=(10, 8))

sns.heatmap(
    correlation,
    annot=True,
    cmap='coolwarm',
    fmt='.2f'
)

plt.title("Correlation Matrix")
plt.show()


# ============================================
# 7. Select Features and Target
# ============================================

features = [
    'pickup_longitude',
    'pickup_latitude',
    'dropoff_longitude',
    'dropoff_latitude',
    'passenger_count',
    'hour',
    'day',
    'month',
    'year',
    'day_of_week',
    'distance_km'
]

X = df[features]
y = df['fare_amount']


# ============================================
# 8. Train-Test Split
# ============================================

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42
)

print("\nTraining data:", X_train.shape)
print("Testing data:", X_test.shape)


# ============================================
# 9. Linear Regression
# ============================================

linear_model = LinearRegression()

linear_model.fit(
    X_train,
    y_train
)

linear_pred = linear_model.predict(X_test)

linear_r2 = r2_score(
    y_test,
    linear_pred
)

linear_rmse = np.sqrt(
    mean_squared_error(
        y_test,
        linear_pred
    )
)

linear_mae = mean_absolute_error(
    y_test,
    linear_pred
)

print("\n========== LINEAR REGRESSION ==========")
print("R2 Score :", linear_r2)
print("RMSE     :", linear_rmse)
print("MAE      :", linear_mae)


# ============================================
# 10. Random Forest Regression
# ============================================

rf_model = RandomForestRegressor(
    n_estimators=100,
    random_state=42,
    n_jobs=-1
)

rf_model.fit(
    X_train,
    y_train
)

rf_pred = rf_model.predict(X_test)

rf_r2 = r2_score(
    y_test,
    rf_pred
)

rf_rmse = np.sqrt(
    mean_squared_error(
        y_test,
        rf_pred
    )
)

rf_mae = mean_absolute_error(
    y_test,
    rf_pred
)

print("\n========== RANDOM FOREST REGRESSION ==========")
print("R2 Score :", rf_r2)
print("RMSE     :", rf_rmse)
print("MAE      :", rf_mae)


# ============================================
# 11. Model Comparison
# ============================================

results = pd.DataFrame({
    'Model': [
        'Linear Regression',
        'Random Forest Regression'
    ],
    'R2 Score': [
        linear_r2,
        rf_r2
    ],
    'RMSE': [
        linear_rmse,
        rf_rmse
    ],
    'MAE': [
        linear_mae,
        rf_mae
    ]
})

print("\n========== MODEL COMPARISON ==========")
print(results)


# ============================================
# 12. Actual vs Predicted - Linear Regression
# ============================================

plt.figure(figsize=(8, 6))

plt.scatter(
    y_test,
    linear_pred,
    alpha=0.3
)

plt.xlabel("Actual Fare")
plt.ylabel("Predicted Fare")
plt.title("Linear Regression - Actual vs Predicted")

plt.show()


# ============================================
# 13. Actual vs Predicted - Random Forest
# ============================================

plt.figure(figsize=(8, 6))

plt.scatter(
    y_test,
    rf_pred,
    alpha=0.3
)

plt.xlabel("Actual Fare")
plt.ylabel("Predicted Fare")
plt.title("Random Forest - Actual vs Predicted")

plt.show()


# ============================================
# 14. Random Forest Feature Importance
# ============================================

importance = pd.DataFrame({
    'Feature': features,
    'Importance': rf_model.feature_importances_
})

importance = importance.sort_values(
    by='Importance',
    ascending=False
)

print("\n========== FEATURE IMPORTANCE ==========")
print(importance)

plt.figure(figsize=(10, 6))

sns.barplot(
    x='Importance',
    y='Feature',
    data=importance
)

plt.title("Random Forest Feature Importance")
plt.show()


# ============================================
# 15. Best Model
# ============================================

if rf_r2 > linear_r2:
    print("\nBest Model: Random Forest Regression")
else:
    print("\nBest Model: Linear Regression")

##1. Outlier Detection (IQR Method)
- Q1: First quartile (25th percentile)
- Q3: Third quartile (75th percentile)
- IQR:\[
  IQR = Q3 - Q1
  \]
- Lower Bound:\[
  Q1 - 1.5 \times IQR
  \]
- Upper Bound:\[
  Q3 + 1.5 \times IQR
  \]
Values outside these bounds are considered outliers.
2. Haversine Formula (Trip Distance)
Used to calculate the straight-line distance between pickup and drop-off coordinates.
\[
a=\sin^2\left(\frac{\Delta lat}{2}\right)+\cos(lat_1)\cos(lat_2)\sin^2\left(\frac{\Delta lon}{2}\right)
\]
\[
c=2\arcsin(\sqrt{a})
\]
\[
\boxed{Distance = 6371 \times c}
\]
- Latitude and longitude must be converted to radians.
- 6371 km is the approximate radius of Earth.
3. Correlation
Measures the relationship between two variables.

Pearson correlation coefficient:
\[
r=\frac{\operatorname{Cov}(X,Y)}{\sigma_X\sigma_Y}
\]
- \(r=+1\): Perfect positive correlation
- \(r=-1\): Perfect negative correlation
- \(r=0\): No linear correlation
4. Linear Regression
Predicts a continuous value using a linear relationship.
\[
\boxed{y = b_0+b_1x_1+b_2x_2+\cdots+b_nx_n}
\]
Where:
- \(y\) = predicted fare
- \(b_0\) = intercept
- \(b_i\) = model coefficients
- \(x_i\) = input features
5. Random Forest Regression
- Combines predictions from multiple decision trees.
- Captures nonlinear relationships.
- Final prediction is the average of individual tree predictions.
\[
\boxed{\hat{y}=\frac{1}{N}\sum_{i=1}^{N}\hat{y}_i}
\]
Here, \(N\) is the number of trees.
6. Model Evaluation Metrics
R² Score
Measures how well the model explains variation in the target.
\[
R^2=1-\frac{\sum(y_i-\hat{y}_i)^2}{\sum(y_i-\bar{y})^2}
\]
Higher R² is generally better.
Mean Squared Error (MSE)
\[
MSE=\frac{1}{n}\sum_{i=1}^{n}(y_i-\hat{y}_i)^2
\]
Root Mean Squared Error (RMSE)
\[
\boxed{RMSE=\sqrt{MSE}}
\]
Mean Absolute Error (MAE)
\[
\boxed{MAE=\frac{1}{n}\sum_{i=1}^{n}|y_i-\hat{y}_i|}
\]
Lower RMSE and MAE indicate better predictions.
Accuracy: Overall proportion of correct predictions.
\[
\boxed{Accuracy=\frac{TP+TN}{TP+TN+FP+FN}}
\]
Error Rate: Proportion of incorrect predictions.
\[
\boxed{Error\ Rate=1-Accuracy}
\]
Precision: Proportion of predicted diabetic cases that are actually diabetic.
\[
\boxed{Precision=\frac{TP}{TP+FP}}
\]
Recall: Proportion of actual diabetic cases correctly identified.
\[
\boxed{Recall=\frac{TP}{TP+FN}}
\]
F1-Score: Harmonic mean of precision and recall.
\[
\boxed{F1=2\times\frac{Precision\times Recall}{Precision+Recall}}
\]
